  { id: 4, title: "Project Four", url: "https://player.vimeo.com/video/000004" },
  { id: 5, title: "Project Five", url: "https://player.vimeo.com/video/000005" },
  { id: 6, title: "Project Six", url: "https://player.vimeo.com/video/000006" },

  { id: 9, title: "Project Nine", url: "https://player.vimeo.com/video/000009" },
  { id: 10, title: "Project Ten", url: "https://player.vimeo.com/video/000010" },
];

function Video({ title, url }) {
  return (
    <div className="mb-6 w-full md:w-1/2 lg:w-1/3 p-2">
      <iframe
        className="w-full aspect-video rounded-lg shadow"
        src={url}
        title={title}
        frameBorder="0"
        allow="autoplay; fullscreen"
        allowFullScreen
      ></iframe>
      <p className="text-sm text-center mt-2">{title}</p>
    </div>
  );
}

function App() {
  const [showAll, setShowAll] = useState(false);
  const [showForm, setShowForm] = useState(false);

  const visibleVideos = showAll ? videos : videos.slice(0, 3);

  return (
    <div className="relative min-h-screen bg-white text-black px-4 py-8">
      <h1 className="text-2xl font-bold text-center mb-8">🎬 Motion Graphics Portfolio</h1>

      <div className="flex flex-wrap justify-center">
        {visibleVideos.map((v) => (
          <Video key={v.id} title={v.title} url={v.url} />
        ))}
      </div>

      {!showAll && (
        <div className="text-center mt-6">
          <button
            onClick={() => setShowAll(true)}
            className="text-blue-600 underline hover:text-blue-800"
          >
            더보기
          </button>
        </div>
      )}

      {/* Floating message icon */}
      <button
        onClick={() => setShowForm(!showForm)}
        className="fixed bottom-6 right-6 w-14 h-14 bg-blue-600 text-white rounded-full shadow-lg flex items-center justify-center text-2xl hover:bg-blue-700"
        title="메시지 남기기"
      >
        💬
      </button>

      {/* Message form */}
      {showForm && (
        <div className="fixed bottom-24 right-6 bg-white border shadow-lg p-4 rounded-lg w-72">
          <h2 className="text-lg font-semibold mb-2">메시지를 남겨주세요</h2>
          <textarea
            rows="4"
            placeholder="문의사항이나 메모를 작성해주세요."
            className="w-full border p-2 rounded mb-2 text-sm"
          ></textarea>
          <button className="bg-blue-600 text-white px-4 py-1 rounded hover:bg-blue-700 text-sm">
            보내기
          </button>
        </div>
      )}
    </div>
  );
}

const root = createRoot(document.getElementById("root"));
root.render(<App />);
